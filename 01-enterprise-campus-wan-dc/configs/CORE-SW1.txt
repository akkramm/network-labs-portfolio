! CORE-SW1 : Catalyst 3560 - Campus Core 1 (HSRP Active, priority 250)
! NOTE: add the remaining sections (Port-channel 6, Fa0/5 uplink to Router1,
! OSPF Area 1 and TACACS+) from "show running-config" before publishing.

enable
configure terminal
hostname CORE-SW1
ip routing

! VLAN Provisioning
vlan 10 name HR
vlan 20 name Finance
vlan 30 name IT-Wireless
vlan 99 name CORE-TRANSIT
vlan 100 name Management
exit

! SVI and HSRP Active Configuration (Priority 250 + Preempt)
interface Vlan10
 ip address 192.168.1.50 255.255.255.128
 standby 10 ip 192.168.1.25
 standby 10 priority 250
 standby 10 preempt
 no shutdown
interface Vlan20
 ip address 192.168.2.50 255.255.255.128
 standby 20 ip 192.168.2.25
 standby 20 priority 250
 standby 20 preempt
 no shutdown
interface Vlan30
 ip address 192.168.3.50 255.255.255.128
 standby 30 ip 192.168.3.25
 standby 30 priority 250
 standby 30 preempt
 no shutdown
interface Vlan99
 ip address 10.99.99.1 255.255.255.252
 no shutdown
interface Vlan100
 ip address 192.168.9.50 255.255.255.128
 standby 100 ip 192.168.9.25
 standby 100 priority 250
 standby 100 preempt
 no shutdown

! Centralized DHCP Server Pools and Infrastructure Exclusions
ip dhcp excluded-address 192.168.1.1 192.168.1.51
ip dhcp excluded-address 192.168.2.1 192.168.2.51
ip dhcp excluded-address 192.168.3.1 192.168.3.51
ip dhcp excluded-address 192.168.3.100
ip dhcp excluded-address 192.168.9.1 192.168.9.51
ip dhcp excluded-address 192.168.9.100

ip dhcp pool HR-VLAN-10
 network 192.168.1.0 255.255.255.128
 default-router 192.168.1.25
 dns-server 8.8.8.8
ip dhcp pool FIN-VLAN-20
 network 192.168.2.0 255.255.255.128
 default-router 192.168.2.25
 dns-server 8.8.8.8
ip dhcp pool IT-VLAN-30
 network 192.168.3.0 255.255.255.128
 default-router 192.168.3.25
 dns-server 8.8.8.8
ip dhcp pool MGMT-VLAN-100
 network 192.168.9.0 255.255.255.128
 default-router 192.168.9.25
 dns-server 8.8.8.8
 option 150 ip 192.168.9.100

! Downlinks to Access Switches (SW1 and SW2)
interface range FastEthernet0/1 - 2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,99,100
 no shutdown

! 2x 1Gbps Inter-Core EtherChannel (Port-Channel 6 - LACP Active)
interface range GigabitEthernet0/1 - 2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,99,100
 channel-group 6 mode active
 no shutdown

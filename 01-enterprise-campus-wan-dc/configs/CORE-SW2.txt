! CORE-SW2 : Catalyst 3560 - Campus Core 2 (HSRP Standby, priority 100)

enable
configure terminal
hostname CORE-SW2
ip routing

vlan 10 name HR
vlan 20 name Finance
vlan 30 name IT-Wireless
vlan 99 name CORE-TRANSIT
vlan 100 name Management
exit

! SVI and HSRP Standby Configuration (Priority 100 + Preempt)
interface Vlan10
 ip address 192.168.1.51 255.255.255.128
 standby 10 ip 192.168.1.25
 standby 10 priority 100
 standby 10 preempt
 no shutdown
interface Vlan20
 ip address 192.168.2.51 255.255.255.128
 standby 20 ip 192.168.2.25
 standby 20 priority 100
 standby 20 preempt
 no shutdown
interface Vlan30
 ip address 192.168.3.51 255.255.255.128
 standby 30 ip 192.168.3.25
 standby 30 priority 100
 standby 30 preempt
 no shutdown
interface Vlan99
 ip address 10.99.99.2 255.255.255.252
 no shutdown
interface Vlan100
 ip address 192.168.9.51 255.255.255.128
 standby 100 ip 192.168.9.25
 standby 100 priority 100
 standby 100 preempt
 no shutdown

! Downlink Trunks to SW1 and SW2
interface range FastEthernet0/1 - 2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,99,100
 no shutdown

! Dedicated Trunk to WLC-2504 (Port 1) - Native VLAN 100 untagged for initial management discovery
interface FastEthernet0/6
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 100
 switchport trunk allowed vlan 2-1005
 no shutdown

! 2x 1Gbps Inter-Core EtherChannel (Port-Channel 6 - LACP Active)
interface range GigabitEthernet0/1 - 2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,99,100
 channel-group 6 mode active
 no shutdown
interface Port-channel 6
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,99,100
 no shutdown

! Routed P2P Uplink to Router1
interface FastEthernet0/5
 no switchport
 ip address 172.168.2.2 255.255.255.252
 no shutdown

! OSPF Area 1
router ospf 1
 router-id 20.20.20.20
 network 172.168.2.0 0.0.0.3 area 1
 network 10.99.99.0 0.0.0.3 area 1
 network 192.168.1.0 0.0.0.127 area 1
 network 192.168.2.0 0.0.0.127 area 1
 network 192.168.3.0 0.0.0.127 area 1
 network 192.168.9.0 0.0.0.127 area 1
